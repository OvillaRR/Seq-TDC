#
# Devuelve:
#
# embedding GRU
# risk_score
#

# risk_score con el head ya entrenado.
# ============================================================

@torch.no_grad()
def extract_account_embeddings(
    model,
    loader,
    device
):

    model.eval()

    embedding_blocks = []
    score_blocks = []


    for batch in loader:

        batch = {

            k:
                v.to(device)

            for k, v
            in batch.items()
        }


        # ====================================================
        # INPUT NUMÉRICO
        # ====================================================

        x_num = batch[
            "x_num"
        ]


        x_static = batch[
            "x_static"
        ]


        lengths = (
            batch[
                "length"
            ]
            .cpu()
        )


        # ====================================================
        # EMBEDDINGS CATEGÓRICOS
        # ====================================================

        emb_list = []


        for name in model.embeddings.keys():

            emb = model.embeddings[
                name
            ](
                batch[
                    name
                ]
            )

            emb_list.append(
                emb
            )


        # ====================================================
        # MISMA ENTRADA UTILIZADA DURANTE ENTRENAMIENTO
        # ====================================================

        x = torch.cat(

            [x_num]
            +
            emb_list,

            dim=-1
        )


        # ====================================================
        # PACK
        # ====================================================

        packed = (
            nn.utils.rnn
            .pack_padded_sequence(

                x,

                lengths,

                batch_first=True,

                enforce_sorted=False
            )
        )


        # ====================================================
        # GRU
        # ====================================================

        _, h_n = model.gru(
            packed
        )


        # ====================================================
        # ÚLTIMA CAPA GRU
        #
        # Ejemplo:
        #
        # hidden_size = 32
        #
        # h.shape = [batch, 32]
        # ====================================================

        h = h_n[-1]


        embedding_blocks.append(

            h.detach()
            .cpu()
            .numpy()
            .astype(np.float32)
        )


        # ====================================================
        # SCORE DEL MODELO
        #
        # Evitamos volver a ejecutar la GRU.
        #
        # Usamos directamente:
        #
        # h_GRU + estáticas -> head
        # ====================================================

        fused = torch.cat(

            [
                h,
                x_static
            ],

            dim=-1
        )


        logits = (

            model.head(
                fused
            )
            .squeeze(1)
        )


        probs = torch.sigmoid(
            logits
        )


        score_blocks.append(

            probs.detach()
            .cpu()
            .numpy()
            .astype(np.float32)
        )


    embeddings = np.concatenate(
        embedding_blocks,
        axis=0
    )


    scores = np.concatenate(
        score_blocks,
        axis=0
    )


    return (
        embeddings,
        scores
    )

@torch.no_grad()
def embed_event_chunk(
    num_sequences,
    cat_sequences,
    model,
    device
):

    if len(num_sequences) == 0:

        return np.empty(
            (
                0,
                model.gru.hidden_size
            ),
            dtype=np.float32
        )


    # ========================================================
    # PADDING
    # ========================================================

    X_num, lengths = pad_num_sequences(

        num_sequences,

        CFG.max_seq_len,

        len(
            SEQ_NUM_COLS
        )
    )


    X_cat = {

        col:
            pad_cat_sequences(

                cat_sequences[
                    col
                ],

                CFG.max_seq_len
            )

        for col
        in CAT_SEQ_COLS
    }


    # ========================================================
    # TENSORES
    # ========================================================

    x_num = torch.tensor(

        X_num,

        dtype=torch.float32,

        device=device
    )


    lengths_t = torch.tensor(

        lengths,

        dtype=torch.long
    )


    emb_list = []


    for col in CAT_SEQ_COLS:

        x_cat = torch.tensor(

            X_cat[
                col
            ],

            dtype=torch.long,

            device=device
        )


        emb = model.embeddings[
            col
        ](
            x_cat
        )


        emb_list.append(
            emb
        )


    # ========================================================
    # INPUT GRU
    # ========================================================

    x = torch.cat(

        [x_num]
        +
        emb_list,

        dim=-1
    )


    packed = (
        nn.utils.rnn
        .pack_padded_sequence(

            x,

            lengths_t,

            batch_first=True,

            enforce_sorted=False
        )
    )


    _, h_n = model.gru(
        packed
    )


    h = h_n[-1]


    return (

        h.detach()
        .cpu()
        .numpy()
        .astype(np.float32)
    )

# 14) CONSTRUIR BANCO DE TRANSACCIONES SANAS
#
# IMPORTANTE:
#
# trx_score debe ser la base transaccional YA PROCESADA,
# es decir debe contener:
#
# amount_log
# delta_days
# day_of_week
# is_weekend
#
# giro_r
# merchant_r
# city_r
# giro_city_r
# giro_amount_bin_r
#
# Igual que la que utilizaste para build_samples().
# ============================================================

def build_healthy_transaction_bank(

    healthy_account_meta,

    trx_score,

    model,

    vocabs,

    device,

    max_events_per_account=20,

    chunk_size=4096

):

    # ========================================================
    # VERIFICAR COLUMNAS
    # ========================================================

    required_cols = (

        [
            "NUM_CUENTA",
            "txn_date",
            "auth_amt"
        ]

        +

        SEQ_NUM_COLS[:-1]

        +

        CAT_SEQ_COLS
    )


    required_cols = list(
        dict.fromkeys(
            required_cols
        )
    )


    missing = [

        c
        for c in required_cols
        if c not in trx_score.columns
    ]


    if missing:

        raise ValueError(
            f"Faltan columnas en trx_score: {missing}"
        )


    # ========================================================
    # TRANSACCIONES AGRUPADAS
    # ========================================================

    grouped_trx = (

        trx_score

        .sort_values(
            [
                "NUM_CUENTA",
                "txn_date"
            ]
        )

        .groupby(
            "NUM_CUENTA",
            sort=False
        )
    )


    # ========================================================
    # BUFFERS
    # ========================================================

    num_buffer = []


    cat_buffer = {

        col: []

        for col
        in CAT_SEQ_COLS
    }


    metadata_rows = []


    embedding_blocks = []


    # ========================================================
    # RECORRER CLIENTES SANOS
    # ========================================================

    for ref_row in healthy_account_meta.itertuples(
        index=False
    ):

        account = str(
            ref_row.NUM_CUENTA
        )


        if account not in grouped_trx.groups:
            continue


        business_date = pd.Timestamp(
            ref_row.business_date
        )


        g = (

            grouped_trx

            .get_group(
                account
            )

            .sort_values(
                "txn_date"
            )

            .copy()
        )


        # ====================================================
        # SÓLO INFORMACIÓN CONOCIDA EN EL CORTE SANO
        # ====================================================

        cutoff_exclusive = (

            business_date.normalize()

            +

            pd.Timedelta(
                days=1
            )
        )


        g = (

            g[
                g[
                    "txn_date"
                ]
                <
                cutoff_exclusive
            ]

            .copy()
        )


        if (
            len(g)
            <
            CFG.min_txn_history
        ):

            continue


        # ====================================================
        # QUÉ EVENTOS VAMOS A GUARDAR
        #
        # Sólo las últimas N trx de cada cuenta.
        # ====================================================

        start_pos = max(

            CFG.min_txn_history - 1,

            len(g)
            -
            max_events_per_account
        )


        # ====================================================
        # CADA TRANSACCIÓN GENERA UN ESTADO GRU
        # ====================================================

        for pos in range(
            start_pos,
            len(g)
        ):

            current_tx = (
                g.iloc[
                    pos
                ]
            )


            event_asof = pd.Timestamp(
                current_tx[
                    "txn_date"
                ]
            )


            # =================================================
            # HISTORIA HASTA ESTA TRANSACCIÓN
            # =================================================

            hist = (

                g.iloc[
                    : pos + 1
                ]

                .tail(
                    CFG.max_seq_len
                )

                .copy()
            )


            # =================================================
            # MUY IMPORTANTE
            #
            # Para monitoreo transacción por transacción:
            #
            # days_to_asof se recalcula contra ESTA
            # transacción, no contra el business_date mensual.
            #
            # La trx actual queda con:
            #
            # days_to_asof = 0
            # =================================================

            hist[
                "days_to_asof"
            ] = (

                (
                    event_asof

                    -

                    hist[
                        "txn_date"
                    ]
                )

                .dt
                .total_seconds()

                /

                86400.0
            )


            hist[
                "days_to_asof"
            ] = (

                hist[
                    "days_to_asof"
                ]

                .clip(
                    lower=0
                )

                .astype(
                    np.float32
                )
            )


            # =================================================
            # SECUENCIA NUMÉRICA
            # =================================================

            seq_num = (

                hist[
                    SEQ_NUM_COLS
                ]

                .to_numpy(
                    dtype=np.float32
                )
            )


            num_buffer.append(
                seq_num
            )


            # =================================================
            # CATEGÓRICAS
            # =================================================

            for col in CAT_SEQ_COLS:

                ids = (

                    hist[
                        col
                    ]

                    .astype(str)

                    .map(

                        lambda x:

                            vocabs[
                                col
                            ].get(

                                x,

                                vocabs[
                                    col
                                ][
                                    "<UNK>"
                                ]
                            )
                    )

                    .to_numpy(
                        dtype=np.int64
                    )
                )


                cat_buffer[
                    col
                ].append(
                    ids
                )


            # =================================================
            # METADATA
            # =================================================

            metadata_rows.append({

                "NUM_CUENTA":
                    account,

                "reference_business_date":
                    business_date,

                "txn_date":
                    current_tx[
                        "txn_date"
                    ],

                "auth_amt":
                    current_tx.get(
                        "auth_amt",
                        np.nan
                    ),

                "giro":
                    current_tx.get(
                        "giro",
                        None
                    ),

                "giro_r":
                    current_tx.get(
                        "giro_r",
                        None
                    ),

                "merchant":
                    current_tx.get(
                        "merchant",
                        None
                    ),

                "merchant_r":
                    current_tx.get(
                        "merchant_r",
                        None
                    ),

                "city":
                    current_tx.get(
                        "city",
                        None
                    ),

                "city_r":
                    current_tx.get(
                        "city_r",
                        None
                    ),

                "healthy_account_cluster":
                    ref_row.cluster,

                "sequence_length":
                    len(hist)
            })


            # =================================================
            # FLUSH DEL CHUNK
            # =================================================

            if (
                len(
                    num_buffer
                )
                >=
                chunk_size
            ):

                emb_chunk = embed_event_chunk(

                    num_buffer,

                    cat_buffer,

                    model,

                    device
                )


                embedding_blocks.append(
                    emb_chunk
                )


                # ---------------------------------------------
                # Vaciar buffers
                # ---------------------------------------------

                num_buffer = []


                cat_buffer = {

                    col: []

                    for col
                    in CAT_SEQ_COLS
                }


    # ========================================================
    # ÚLTIMO CHUNK
    # ========================================================

    if len(
        num_buffer
    ) > 0:

        emb_chunk = embed_event_chunk(

            num_buffer,

            cat_buffer,

            model,

            device
        )


        embedding_blocks.append(
            emb_chunk
        )


    # ========================================================
    # CONCATENAR
    # ========================================================

    if len(
        embedding_blocks
    ) == 0:

        raise ValueError(
            "No se generaron embeddings transaccionales."
        )


    embeddings = np.concatenate(

        embedding_blocks,

        axis=0
    )


    metadata = pd.DataFrame(
        metadata_rows
    )


    if (
        len(metadata)
        !=
        len(embeddings)
    ):

        raise ValueError(
            "Metadata y embeddings transaccionales "
            "quedaron desalineados."
        )


    return (
        embeddings,
        metadata
    )

def project_new_accounts(

    X_num_new,

    X_cat_new,

    X_static_new,

    len_new,

    keys_new,

    model,

    reference_bank,

    device,

    batch_size=512

):

    # ========================================================
    # TARGET DUMMY
    # ========================================================

    y_dummy = np.zeros(

        len(
            X_num_new
        ),

        dtype=np.float32
    )


    # ========================================================
    # DATASET
    # ========================================================

    dataset = AccountSequenceDataset(

        X_num_new,

        X_cat_new,

        X_static_new,

        y_dummy,

        len_new
    )


    loader = DataLoader(

        dataset,

        batch_size=
            batch_size,

        shuffle=False
    )


    # ========================================================
    # GRU PROJECTION
    # ========================================================

    embeddings, scores = (
        extract_account_embeddings(

            model,

            loader,

            device
        )
    )


    # ========================================================
    # MISMO SCALER SANO
    # ========================================================

    Z = (

        reference_bank[
            "account_scaler"
        ]

        .transform(
            embeddings
        )

        .astype(
            np.float32
        )
    )


    # ========================================================
    # ASIGNACIÓN CLUSTER
    # ========================================================

    clusters = (

        reference_bank[
            "account_kmeans"
        ]

        .predict(
            Z
        )
    )


    # ========================================================
    # DISTANCIA AL CENTRO
    # ========================================================

    centers = (

        reference_bank[
            "account_kmeans"
        ]

        .cluster_centers_[
            clusters
        ]
    )


    distances = np.linalg.norm(

        Z
        -
        centers,

        axis=1
    )


    # ========================================================
    # TABLA
    # ========================================================

    projection = pd.DataFrame(

        keys_new,

        columns=[
            "NUM_CUENTA",
            "business_date"
        ]
    )


    projection[
        "risk_score"
    ] = scores


    projection[
        "healthy_cluster"
    ] = clusters


    projection[
        "distance_to_healthy_centroid"
    ] = distances


    # ========================================================
    # COMPARAR CONTRA P95 DEL CLUSTER SANO
    # ========================================================

    stats = (

        reference_bank[
            "account_stats"
        ][
            [
                "cluster",
                "p95",
                "p99"
            ]
        ]

        .rename(
            columns={

                "cluster":
                    "healthy_cluster",

                "p95":
                    "healthy_distance_p95",

                "p99":
                    "healthy_distance_p99"
            }
        )
    )


    projection = projection.merge(

        stats,

        on="healthy_cluster",

        how="left"
    )


    projection[
        "outside_healthy_p95"
    ] = (

        projection[
            "distance_to_healthy_centroid"
        ]

        >

        projection[
            "healthy_distance_p95"
        ]
    ).astype(int)


    projection[
        "outside_healthy_p99"
    ] = (

        projection[
            "distance_to_healthy_centroid"
        ]

        >

        projection[
            "healthy_distance_p99"
        ]
    ).astype(int)


    # ========================================================
    # RATIO DISTANCIA / P95
    #
    # > 1 significa que está fuera del 95% típico
    # de ese cluster sano.
    # ========================================================

    projection[
        "distance_vs_p95"
    ] = (

        projection[
            "distance_to_healthy_centroid"
        ]

        /

        projection[
            "healthy_distance_p95"
        ]
    )


    return (
        embeddings,
        projection
    )

def find_similar_healthy_accounts(

    query_idx,

    new_embeddings,

    new_projection,

    reference_bank,

    k=10

):

    if query_idx < 0:

        query_idx = (
            len(
                new_embeddings
            )
            +
            query_idx
        )


    query = (

        new_embeddings[
            query_idx
        ]

        .reshape(
            1,
            -1
        )
    )


    n_neighbors = min(

        k,

        len(
            reference_bank[
                "account_meta"
            ]
        )
    )


    distances, indices = (

        reference_bank[
            "nn_accounts"
        ]

        .kneighbors(

            query,

            n_neighbors=
                n_neighbors
        )
    )


    neighbors = (

        reference_bank[
            "account_meta"
        ]

        .iloc[
            indices[0]
        ]

        .copy()

        .reset_index(
            drop=True
        )
    )


    neighbors[
        "cosine_distance"
    ] = distances[0]


    neighbors[
        "similarity"
    ] = (

        1.0

        -

        distances[0]
    )


    query_info = (

        new_projection

        .iloc[
            [
                query_idx
            ]
        ]

        .copy()
    )


    return (
        query_info,
        neighbors
    )

def project_new_transaction_states(

    trx_new,

    model,

    vocabs,

    device,

    max_events_per_account=None,

    chunk_size=2048

):

    grouped_trx = (

        trx_new

        .sort_values(
            [
                "NUM_CUENTA",
                "txn_date"
            ]
        )

        .groupby(
            "NUM_CUENTA",
            sort=False
        )
    )


    num_buffer = []


    cat_buffer = {

        col: []

        for col
        in CAT_SEQ_COLS
    }


    metadata_rows = []


    embedding_blocks = []


    # ========================================================
    # CLIENTES
    # ========================================================

    for account, g in grouped_trx:

        g = (

            g

            .sort_values(
                "txn_date"
            )

            .reset_index(
                drop=True
            )
        )


        if (
            len(g)
            <
            CFG.min_txn_history
        ):

            continue


        # ----------------------------------------------------
        # Todos los eventos o últimas N transacciones
        # ----------------------------------------------------

        if (
            max_events_per_account
            is None
        ):

            start_pos = (
                CFG.min_txn_history
                -
                1
            )

        else:

            start_pos = max(

                CFG.min_txn_history
                -
                1,

                len(g)
                -
                max_events_per_account
            )


        # ====================================================
        # CADA TRANSACCIÓN
        # ====================================================

        for pos in range(
            start_pos,
            len(g)
        ):

            current_tx = (
                g.iloc[
                    pos
                ]
            )


            event_asof = pd.Timestamp(
                current_tx[
                    "txn_date"
                ]
            )


            # =================================================
            # SECUENCIA HASTA ESTE EVENTO
            # =================================================

            hist = (

                g.iloc[
                    : pos + 1
                ]

                .tail(
                    CFG.max_seq_len
                )

                .copy()
            )


            # =================================================
            # RECENCIA RELATIVA A LA TRANSACCIÓN ACTUAL
            # =================================================

            hist[
                "days_to_asof"
            ] = (

                (
                    event_asof

                    -

                    hist[
                        "txn_date"
                    ]
                )

                .dt
                .total_seconds()

                /

                86400.0
            )


            hist[
                "days_to_asof"
            ] = (

                hist[
                    "days_to_asof"
                ]

                .clip(
                    lower=0
                )

                .astype(
                    np.float32
                )
            )


            # =================================================
            # NUMÉRICAS
            # =================================================

            seq_num = (

                hist[
                    SEQ_NUM_COLS
                ]

                .to_numpy(
                    dtype=np.float32
                )
            )


            num_buffer.append(
                seq_num
            )


            # =================================================
            # CATEGÓRICAS
            # =================================================

            for col in CAT_SEQ_COLS:

                ids = (

                    hist[
                        col
                    ]

                    .astype(str)

                    .map(

                        lambda x:

                            vocabs[
                                col
                            ].get(

                                x,

                                vocabs[
                                    col
                                ][
                                    "<UNK>"
                                ]
                            )
                    )

                    .to_numpy(
                        dtype=np.int64
                    )
                )


                cat_buffer[
                    col
                ].append(
                    ids
                )


            # =================================================
            # METADATA DEL EVENTO
            # =================================================

            metadata_rows.append({

                "NUM_CUENTA":
                    account,

                "txn_number":
                    pos + 1,

                "txn_date":
                    current_tx[
                        "txn_date"
                    ],

                "auth_amt":
                    current_tx.get(
                        "auth_amt",
                        np.nan
                    ),

                "giro":
                    current_tx.get(
                        "giro",
                        None
                    ),

                "giro_r":
                    current_tx.get(
                        "giro_r",
                        None
                    ),

                "merchant":
                    current_tx.get(
                        "merchant",
                        None
                    ),

                "merchant_r":
                    current_tx.get(
                        "merchant_r",
                        None
                    ),

                "city":
                    current_tx.get(
                        "city",
                        None
                    ),

                "city_r":
                    current_tx.get(
                        "city_r",
                        None
                    ),

                "sequence_length":
                    len(hist)
            })


            # =================================================
            # CHUNK
            # =================================================

            if (
                len(
                    num_buffer
                )
                >=
                chunk_size
            ):

                emb_chunk = (
                    embed_event_chunk(

                        num_buffer,

                        cat_buffer,

                        model,

                        device
                    )
                )


                embedding_blocks.append(
                    emb_chunk
                )


                num_buffer = []


                cat_buffer = {

                    col: []

                    for col
                    in CAT_SEQ_COLS
                }


    # ========================================================
    # ÚLTIMO BLOQUE
    # ========================================================

    if len(
        num_buffer
    ) > 0:

        emb_chunk = embed_event_chunk(

            num_buffer,

            cat_buffer,

            model,

            device
        )


        embedding_blocks.append(
            emb_chunk
        )


    if len(
        embedding_blocks
    ) == 0:

        raise ValueError(
            "No se generaron estados transaccionales."
        )


    embeddings = np.concatenate(

        embedding_blocks,

        axis=0
    )


    metadata = pd.DataFrame(
        metadata_rows
    )


    return (
        embeddings,
        metadata
    )

        result[
            "distance_to_healthy_tx_centroid"
        ]

        /

        result[
            "healthy_tx_distance_p95"
        ]
    )


    return result

# ============================================================


def find_similar_healthy_transactions(

    query_idx,

    new_tx_embeddings,

    new_tx_projection,

    reference_bank,

    k=10

):

    # --------------------------------------------------------
    # Permitir query_idx = -1
    # para última transacción.
    # --------------------------------------------------------

    if query_idx < 0:

        query_idx = (

            len(
                new_tx_embeddings
            )

            +

            query_idx
        )


    query = (

        new_tx_embeddings[
            query_idx
        ]

        .reshape(
            1,
            -1
        )
    )


    n_neighbors = min(

        k,

        len(
            reference_bank[
                "tx_meta"
            ]
        )
    )


    distances, indices = (

        reference_bank[
            "nn_transactions"
        ]

        .kneighbors(

            query,

            n_neighbors=
                n_neighbors
        )
    )


    neighbors = (

        reference_bank[
            "tx_meta"
        ]

        .iloc[
            indices[0]
        ]

        .copy()

        .reset_index(
            drop=True
        )
    )


    neighbors[
        "cosine_distance"
    ] = distances[0]


    neighbors[
        "similarity"
    ] = (

        1.0

        -

        distances[0]
    )


    query_info = (

        new_tx_projection

        .iloc[
            [
                query_idx
            ]
        ]

        .copy()
    )


    return (
        query_info,
        neighbors
    )
